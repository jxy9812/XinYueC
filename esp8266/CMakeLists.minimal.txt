# 最小二分应用：同一链接脚本/工具链/烧录布局，但不含 XinYueC。
cmake_minimum_required(VERSION 3.16)
project(xinyuec_minimal C)

set(CMAKE_C_STANDARD 99)

set(XESP8266_SDK_PATH "" CACHE PATH "ESP8266 RTOS SDK")
set(XESP8266_TOOLCHAIN_ROOT "" CACHE PATH "xtensa toolchain")
if(NOT XESP8266_SDK_PATH)
    message(FATAL_ERROR "need -DXESP8266_SDK_PATH")
endif()

set(SDK_INCLUDE_DIRS
    "${XESP8266_SDK_PATH}/include"
    "${XESP8266_SDK_PATH}/include/espressif"
    "${XESP8266_SDK_PATH}/include/freertos"
    "${XESP8266_SDK_PATH}/include/espressif/esp8266"
    "${XESP8266_SDK_PATH}/include/lwip"
    "${XESP8266_SDK_PATH}/include/lwip/ipv4"
    "${XESP8266_SDK_PATH}/include/lwip/ipv6"
    "${XESP8266_SDK_PATH}/include/espconn"
    "${XESP8266_SDK_PATH}/extra_include"
    "${XESP8266_SDK_PATH}/driver_lib/include")

set(SDK_LIBS
    cirom crypto espconn espnow freertos gcc
    json hal lwip main mesh mirom net80211 nopoll mbedtls
    c gcc
    phy pp pwm smartconfig spiffs ssl wpa wps)

get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(LDSCRIPT "${CMAKE_CURRENT_LIST_DIR}/ld/eagle.app.v6.4m.ld")

add_compile_definitions(__ESP8266__ __FreeRTOS__ __ets__ ICACHE_FLASH F_CPU=80000000)
add_compile_options(-Os -std=gnu99 -U__STRICT_ANSI__ -ffunction-sections -fdata-sections -Wno-error)

add_executable(minimal ${CMAKE_CURRENT_LIST_DIR}/user/minimal_main.c)
target_include_directories(minimal PRIVATE ${SDK_INCLUDE_DIRS})
target_link_directories(minimal PRIVATE
    "${XESP8266_SDK_PATH}/lib"
    "${XESP8266_TOOLCHAIN_ROOT}/lib/gcc/xtensa-lx106-elf/10.3.0")
target_link_libraries(minimal PRIVATE
    "-Wl,--start-group" ${SDK_LIBS} "-Wl,--end-group"
    "${XESP8266_TOOLCHAIN_ROOT}/lib/gcc/xtensa-lx106-elf/10.3.0/libgcc.a")
target_link_options(minimal PRIVATE
    -B${XESP8266_TOOLCHAIN_ROOT}/bin
    -nostdlib -Wl,--no-check-sections -Wl,-static -Wl,--gc-sections
    "SHELL:-T ${LDSCRIPT}"
    -u call_user_start)
set_target_properties(minimal PROPERTIES SUFFIX ".elf")
